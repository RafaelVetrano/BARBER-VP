  calcComissaoBarbeiro(b, rule) {
    let pctServicos, regraLabel;
    if (rule.tipo === 'percentual') {
      pctServicos = parseFloat(rule.pctServicos) || 0;
      regraLabel = `Percentual fixo ${pctServicos}%`;
    } else {
      const faixaAscending = rule.faixas.filter(f=>!f.acima).sort((a,b2)=>parseFloat(a.ate)-parseFloat(b2.ate));
      const acima = rule.faixas.find(f=>f.acima);
      let matched = faixaAscending.find(f => b.faturadoServicos <= parseFloat(f.ate));
      if (!matched) matched = acima || faixaAscending[faixaAscending.length-1];
      pctServicos = parseFloat(matched && matched.pct) || 0;
      regraLabel = `Faixa ${pctServicos}%`;
    }
    const pctProdutos = parseFloat(rule.pctProdutos) || 0;
    const comissaoServicos = b.faturadoServicos * pctServicos / 100;
    const comissaoProdutos = b.faturadoProdutos * pctProdutos / 100;
    const valeAplicado = rule.descontarVales ? b.vale : 0;
    const total = comissaoServicos + comissaoProdutos - valeAplicado;
    return { pctServicos, regraLabel, comissaoServicos, comissaoProdutos, valeAplicado, total };
  }

  buildChartGeometry(values, w, h) {
    const max = Math.max(...values) * 1.15;
    const min = 0;
    const n = values.length;
    const stepX = w / (n - 1);
    const points = values.map((v, i) => ({
      x: i * stepX,
      y: h - ((v - min) / (max - min)) * (h - 20) - 6,
      value: v,
    }));
    const linePath = points.map((p, i) => (i === 0 ? 'M' : 'L') + p.x.toFixed(1) + ',' + p.y.toFixed(1)).join(' ');
    const areaPath = linePath + ` L${w},${h} L0,${h} Z`;
    const goalPerPoint = 28000 / 30;
    const goalY = h - ((goalPerPoint - min) / (max - min)) * (h - 20) - 6;
    return {points, linePath, areaPath, goalY};
  }

    const prevRevenue = totalRevenue * 0.9;
    const deltaPct = Math.round(((totalRevenue - prevRevenue) / prevRevenue) * 100);

    const maxRevService = Math.max(...prod.serviceRows.map(s=>s.revenue), 1);
    const revServiceRows = prod.serviceRows.map(s => ({
      name: s.name, revenueFmt: this.fmtBRL(s.revenue),
      barStyle: {height:'10px', width:(s.revenue/maxRevService*100)+'%', background:'#D4A84C', borderRadius:'5px'},
    }));

    const PAYMENT_SEGMENTS = [
      {label:'Pix', pct:48, color:'#D4A84C'}, {label:'Crédito', pct:22, color:'#5B8DE0'},
      {label:'Dinheiro', pct:18, color:'#3FB68B'}, {label:'Débito', pct:12, color:'#E8A13C'},
    ];
    const R = 54, CIRC = 2*Math.PI*R;
    let acc = 0;
    const donutSegments = PAYMENT_SEGMENTS.map(s => {
      const dash = (s.pct/100)*CIRC;
      const seg = {color:s.color, dasharray: `${dash.toFixed(2)} ${(CIRC-dash).toFixed(2)}`, dashoffset: (-acc).toFixed(2)};
      acc += dash;
      return seg;
    });
    const paymentLegend = PAYMENT_SEGMENTS.map(s => ({label:s.label, color:s.color, pctLabel: s.pct+'%'}));

    const atendimentos = Math.max(1, Math.round(prod.comissaoData.faturadoServicos / 62));
    const ticketMedioFmt = this.fmtBRL(prod.comissaoData.faturadoServicos / atendimentos);

    return {
      isRelatorios, periodPills, isRelCustom, relCustomFrom, relCustomTo,
      updRelCustomFrom: (e) => this.setState({relCustomFrom: e.target.value}),
      updRelCustomTo: (e) => this.setState({relCustomTo: e.target.value}),
      exportRelatorioPdf: () => this.showToast('Relatório exportado em PDF'),
      exportRelatorioCsv: () => this.showToast('Relatório exportado em CSV'),
      totalRevenueFmt: this.fmtBRL(totalRevenue),
      deltaPctLabel: (deltaPct>=0?'+':'') + deltaPct + '% vs. período anterior',
      deltaColor: deltaPct>=0?'#3FB68B':'#E05B5B',
      revAreaPath: revGeo.areaPath, revLinePath: revGeo.linePath, revW: REV_W, revH: REV_H,
      revXLabels, revYLabels,
      revServiceRows,
      donutSegments, paymentLegend, donutR: R,
      atendimentosLabel: atendimentos + ' atendimentos no período',
      ticketMedioFmt,
    };
  }

  renderVals  renderVals() {
    const barberLogado = this.getBarberLogado();
    const { collapsed, activeKey, openMenu, apptMenuOpenIdx, chartPeriod, hoveredPointIdx } = this.state;
    const isDashboard = activeKey === 'dashboard';
    const showEmptyState = false;

    const navItems = NAV_DEFS.map(n => {
      const active = n.key === activeKey;
      return {
        key: n.key, label: n.label, iconPath: n.path, badge: false, locked: false,
        iconColor: active ? '#D4A84C' : '#9AA1AC',
        onClick: () => this.setActive(n.key),
        rowStyle: {
          display: 'flex', alignItems: 'center', gap: '12px',
          height: '40px', padding: collapsed ? '0' : '0 12px',
          justifyContent: collapsed ? 'center' : 'flex-start',
          borderRadius: '9px', cursor: 'pointer',
          background: active ? 'rgba(212,168,76,0.14)' : 'transparent',
        },
      };
    });

    const teamData = TEAM_DATA.find(b => b.id === barberLogado.id) || TEAM_DATA[0];
    const ownAppts = APPOINTMENTS_DAY.filter(a => a.barberId === barberLogado.id);
    const concluidos = ownAppts.filter(a => a.status === 'Concluído').length;
    const fmtHM = (mins) => Math.floor(mins/60).toString().padStart(2,'0') + ':' + (mins%60).toString().padStart(2,'0');
    const now = new Date();
    const nowMins = now.getHours()*60 + now.getMinutes();
    const proximo = ownAppts.filter(a => a.status !== 'Concluído' && a.status !== 'Cancelado' && a.start >= nowMins).sort((a,b)=>a.start-b.start)[0] || null;

    const activeDay = teamData.schedule.find(d => d.active);
    const parseHM = (t) => { const [h,m] = t.split(':').map(Number); return h*60+m; };
    const workMinutes = activeDay ? (parseHM(activeDay.end) - parseHM(activeDay.start) - 60) : 480;
    const bookedMinutes = ownAppts.reduce((s,a)=>s+a.duration, 0);
    const ocupacaoPct = Math.max(0, Math.min(100, Math.round((bookedMinutes / workMinutes) * 100)));

    const prod = this.getBarberProdData(barberLogado);
    const barberRule = this.state.barberRules[barberLogado.name];
    const comissaoEstimadaFmt = barberRule ? this.fmtBRL(this.calcComissaoBarbeiro(prod.comissaoData, barberRule).total) : this.fmtBRL(0);

    const CHART_W = 600, CHART_H = 220;
    const chartValues = prod.values30d;
    const geo = this.buildChartGeometry(chartValues, CHART_W, CHART_H);
    const hoverPoints = geo.points.map((p, i) => ({
      idx: i,
      leftPct: (p.x / CHART_W) * 100,
      topPct: (p.y / CHART_H) * 100,
      onEnter: () => this.setHoveredPoint(i),
      zoneStyle: {position:'absolute', left: (p.x/CHART_W)*100 + '%', top:0, width: (100/chartValues.length) + '%', height:'100%', transform:'translateX(-50%)', cursor:'crosshair'},
    }));
    const hovered = hoveredPointIdx != null ? {
      label: 'Dia ' + (hoveredPointIdx+1),
      value: this.fmtBRL(chartValues[hoveredPointIdx]),
      style: {position:'absolute', left: hoverPoints[hoveredPointIdx].leftPct + '%', top: hoverPoints[hoveredPointIdx].topPct + '%', transform:'translate(-50%,-130%)'},
    } : null;
    const periodBtnStyle = (p) => ({
      padding:'6px 12px', borderRadius:'7px', cursor:'pointer',
      font:'600 12px Inter,sans-serif',
      background: chartPeriod === p ? '#D4A84C' : 'transparent',
      color: chartPeriod === p ? '#0F1115' : '#9AA1AC',
    });

    const maxSvc = Math.max(...prod.serviceRows.map(s=>s.revenue), 1);
    const meuServiceRows = prod.serviceRows.map(s => ({
      name: s.name, revenueFmt: this.fmtBRL(s.revenue),
      barStyle: {height:'10px', width: (s.revenue/maxSvc*100) + '%', background:'#D4A84C', borderRadius:'5px'},
    }));

    const STATUS_COLORS = {'Confirmado':'#3FB68B','Pendente':'#E8A13C','Concluído':'#5B8DE0','Falta':'#E05B5B','Cancelado':'#9AA1AC'};
    const appointments = ownAppts.map((a, i) => ({
      time: fmtHM(a.start), client: a.client, service: a.service,
      statusColor: STATUS_COLORS[a.status], status: a.status,
      cancelled: a.status === 'Cancelado',
      textDecoration: a.status === 'Cancelado' ? 'line-through' : 'none',
      menuOpen: apptMenuOpenIdx === i,
      toggleMenu: () => this.toggleApptMenu(i),
    }));

    return {
      isDashboard,
      showEmptyState,
      barberLogadoNome: barberLogado.name,
      kpiAtendimentosCount: ownAppts.length,
      kpiConcluidosLabel: concluidos + (concluidos === 1 ? ' concluído' : ' concluídos'),
      hasProximoCliente: !!proximo,
      proximoClienteNome: proximo ? proximo.client : '',
      proximoClienteInfo: proximo ? fmtHM(proximo.start) + ' · ' + proximo.service : 'Nenhum agendamento restante hoje',
      kpiOcupacaoPct: ocupacaoPct,
      comissaoEstimadaFmt,
      chartPeriod,
      periodDiaStyle: periodBtnStyle('dia'),
      periodSemanaStyle: periodBtnStyle('semana'),
      periodMesStyle: periodBtnStyle('mes'),
      setPeriodDia: () => this.setChartPeriod('dia'),
      setPeriodSemana: () => this.setChartPeriod('semana'),
      setPeriodMes: () => this.setChartPeriod('mes'),
      chartAreaPath: geo.areaPath,
      chartLinePath: geo.linePath,
      chartW: CHART_W, chartH: CHART_H,
      hoverPoints,
      hovered,
      clearHoveredPoint: this.clearHoveredPoint,
      meuServiceRows,
      appointments,
      anyMenuOpenAppt: apptMenuOpenIdx != null,
      ...this.renderAgendaVals(),
      ...this.renderClientesVals(),
      ...this.renderComandasVals(),
      ...this.renderComissoesVals(),
      ...this.renderFidelidadeVals(),
      ...this.renderRelatoriosVals(),
      collapsed: collapsed || this.state.isNarrow,
      isNarrow: this.state.isNarrow,
      sidebarStyle: {
        width: (collapsed || this.state.isNarrow) ? '64px' : '240px', minWidth: (collapsed || this.state.isNarrow) ? '64px' : '240px',
        background: '#181B21', borderRight: '1px solid #2A2F38',
        display: 'flex', flexDirection: 'column', height: '100%',
        transition: 'width 0.18s ease, min-width 0.18s ease', flexShrink: 0,
      },
      headerSearchWrapStyle: { flex: this.state.isNarrow ? '0 0 auto' : 1, display: this.state.isNarrow ? 'none' : 'flex', justifyContent: 'center', minWidth: 0, overflow: 'hidden' },
      newApptBtnStyle: {
        height: '40px', padding: this.state.isNarrow ? '0 10px' : '0 16px', borderRadius: '10px', background: '#D4A84C',
        border: 'none', color: '#0F1115', font: '600 14px Inter,sans-serif', cursor: 'pointer',
        display: 'flex', alignItems: 'center', gap: '6px', whiteSpace: 'nowrap',
      },
      collapseBtnStyle: {
        width: '28px', height: '28px', borderRadius: '7px', display: 'flex',
        alignItems: 'center', justifyContent: 'center', cursor: 'pointer', flexShrink: 0,
        background: 'transparent',
      },
      collapseIconStyle: { color: '#9AA1AC', transform: collapsed ? 'rotate(180deg)' : 'none', transition: 'transform 0.18s ease' },
      navItems,
      toggleCollapse: this.toggleCollapse,
      selectedUnit: this.state.selectedUnit,
      barberInitials: barberLogado.initials,
      barberColor: teamData.color,
      notifOpen: openMenu === 'notif',
      toggleNotifMenu: () => this.toggleMenu('notif'),
      notifBtnBg: openMenu === 'notif' ? '#1F232B' : 'transparent',
      notifications: [
        {text: 'João confirmou o horário de 14h'},
        {text: 'Rafael cancelou o agendamento das 16h'},
        {text: 'Bruno remarcou para sexta-feira'},
      ],
      avatarOpen: openMenu === 'avatar',
      toggleAvatarMenu: () => this.toggleMenu('avatar'),
      anyMenuOpen: !!openMenu || apptMenuOpenIdx != null,
      closeMenus: this.closeMenus,
    };
  }
